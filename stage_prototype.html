// Cloudflare Worker: sfh-stage-relay (Free Plan: 100,000 requests/day)
// Env Secrets: CF_APP_ID, CF_APP_SECRET (from Cloudflare Dashboard -> Realtime -> Calls)
export default {
  async fetch(request, env) {
    const cors = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
      'Access-Control-Allow-Methods': 'POST, PUT, OPTIONS'
    };
    if (request.method === 'OPTIONS') return new Response(null, { headers: cors });

    const url = new URL(request.url);
    const cfPath = url.pathname.replace(/^\/sfh-rtc/, '');
    const cfUrl = `https://rtc.live.cloudflare.com/v1/apps/${env.CF_APP_ID}${cfPath}`;

    const upstream = await fetch(cfUrl, {
      method: request.method,
      headers: {
        'Authorization': `Bearer ${env.CF_APP_SECRET}`,
        'Content-Type': 'application/json'
      },
      body: await request.text()
    });
    return new Response(await upstream.text(), {
      status: upstream.status,
      headers: { ...cors, 'Content-Type': 'application/json' }
    });
  }
};
